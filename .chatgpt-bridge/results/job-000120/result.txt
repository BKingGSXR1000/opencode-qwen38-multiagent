WAKE_ID="$(
python3 - "$MESSAGE" <<'PY'
import hashlib, os, sys, time
msg=sys.argv[1]
stamp=str(time.time_ns())
suffix=hashlib.sha256(
    (stamp+"\0"+msg+"\0"+str(os.getpid())).encode()
).hexdigest()[:16]
print(f"{stamp}-{suffix}")
PY
)"

    "id":wake_id,
    "message":msg,
}, ensure_ascii=False, sort_keys=True)+"\n")

        exit 0

done
