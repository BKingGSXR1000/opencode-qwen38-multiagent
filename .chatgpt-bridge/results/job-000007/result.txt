#!/usr/bin/env bash
set -euo pipefail

TITLE='AI LLMs - Continue Canary Debugging — Mozilla Firefox'

if [[ $# -lt 1 ]]; then
    echo "Usage: $0 'message to send'"
    exit 2
fi

MESSAGE="$*"

mapfile -t WINDOWS < <(
    xdotool search --onlyvisible --name "^${TITLE}$" 2>/dev/null || true
)

if [[ ${#WINDOWS[@]} -ne 1 ]]; then
    echo "ERROR: expected exactly one ChatGPT Firefox window, found ${#WINDOWS[@]}"
    printf '  %s\n' "${WINDOWS[@]:-}"
    exit 1
fi

WIN="${WINDOWS[0]}"

ACTUAL_TITLE="$(xdotool getwindowname "$WIN" 2>/dev/null || true)"
if [[ "$ACTUAL_TITLE" != "$TITLE" ]]; then
    echo "ERROR: window title changed:"
    echo "  $ACTUAL_TITLE"
    exit 1
fi

# Put exact text in clipboard; avoids keyboard-layout problems.
printf '%s' "$MESSAGE" | xclip -selection clipboard

# Bring only the designated ChatGPT window forward.
xdotool windowactivate --sync "$WIN"
sleep 0.4

# Determine window geometry.
eval "$(
    xdotool getwindowgeometry --shell "$WIN" \
      | grep -E '^(WIDTH|HEIGHT)='
)"

# Click safely inside the ChatGPT composer near the bottom-center.
CLICK_X=$(( WIDTH * 55 / 100 ))
CLICK_Y=$(( HEIGHT - 90 ))

xdotool mousemove --window "$WIN" "$CLICK_X" "$CLICK_Y"
xdotool click 1
sleep 0.2

# Paste exact message.
xdotool key --clearmodifiers ctrl+v
sleep 0.2

# Send.
xdotool key --clearmodifiers Return

echo "SENT: $MESSAGE"
