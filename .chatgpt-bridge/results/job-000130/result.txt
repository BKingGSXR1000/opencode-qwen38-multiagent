   703	def max_step_terminal_summary_db(sid):
   704	    """Return a max-step terminal summary even if text/compaction followed it.
   705	
   706	    OpenCode may append compaction summaries or continuation prose after either
   707	    supported max-step terminal marker. Accept that history only when no later
   708	    tool part exists after the marker; any later tool execution means the
   709	    session was no longer terminal at that point.
   710	    """
   711	    latest=last_assistant_text_db(sid)
   712	    marker_re=MAX_STEP_TERMINAL_RE
   713	    if marker_re.search(latest or ""):
   714	        return latest
   715	
   716	    if not v1_runtime_enabled():
   717	        return ""
   718	
   719	    try:
   720	        con=db_connect()
   721	        rows=con.execute(
   722	            "SELECT time_created,data FROM part "
   723	            "WHERE session_id=? ORDER BY time_created,id",

   725	        ).fetchall()
   726	        con.close()
   727	    except Exception:
   728	        return ""

   730	    marker_time=-1
   731	    marker_text=""
   732	    later_tool_time=-1
   733	    for created,raw in rows:

   741	        if part.get("type")=="text":
   742	            text=str(part.get("text") or "")

   744	                marker_time=when
   745	                marker_text=text.strip()
   746	                later_tool_time=-1

   749	            marker_time>=0

   752	        ):

   755	    if marker_time<0 or later_tool_time>marker_time:
   756	        return ""
