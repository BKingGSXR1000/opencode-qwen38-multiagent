def sh(cmd,cwd=None,check=True,timeout=90):
    p=subprocess.run(
        cmd,cwd=str(cwd) if cwd else None,
        text=True,capture_output=True,timeout=timeout
    )
    if check and p.returncode:
        raise RuntimeError(
            f"command failed rc={p.returncode}: {cmd!r}\n"
            f"stdout:\n{p.stdout}\nstderr:\n{p.stderr}"
        )
    return p

def install_job(payload):
    if payload.get("protocol")!="ai-beast-chat-job-v1":
        raise ValueError("invalid protocol")

    job_id=str(payload.get("id") or "")
    if not JOB_RE.fullmatch(job_id):
        raise ValueError("invalid job id")

    timeout=int(payload.get("timeout_seconds",300))
    if timeout < 1 or timeout > 1800:
        raise ValueError("timeout outside 1..1800")

    encoded=payload.get("script_b64")
    digest=str(payload.get("sha256") or "").lower()
    if not isinstance(encoded,str) or not encoded:
        raise ValueError("script_b64 missing")
    if not re.fullmatch(r"[0-9a-f]{64}",digest):
        raise ValueError("sha256 invalid")

        script=base64.b64decode(encoded,validate=True)

        raise ValueError(f"script_b64 invalid: {exc}") from exc

        raise ValueError("script too large")
    if hashlib.sha256(script).hexdigest()!=digest:
        raise ValueError("sha256 mismatch")

    text=script.decode("utf-8")
    if re.search(r"(?m)^[ \t]*sudo(?:[ \t]|$)",text):
        raise ValueError("sudo is forbidden")
    if "\x00" in text:
