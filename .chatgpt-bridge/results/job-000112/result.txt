def remote_result_exists(job_id):
    return result_dir(job_id).is_dir() and (result_dir(job_id) / "COMPLETE").is_file()

def push_spool(job_id, cfg):
    branch = cfg["automation_branch"]

    # Always start from current remote branch. This makes each retry independent
    # of previous failed git state.
    sync_branch(cfg)

    if remote_result_exists(job_id):
        commit = git(
            "log","-1","--format=%H","--",
            str(result_dir(job_id).relative_to(WORK))
        ).stdout.strip()
        return commit

    copy_spool_to_worktree(job_id)

    rel = str(result_dir(job_id).relative_to(WORK))
    git("add","--",rel)

    staged = git("diff","--cached","--quiet",check=False)
    if staged.returncode == 0:
        raise RuntimeError(f"no staged result for {job_id}")

    rc = int((spool_dir(job_id) / "exit-code.txt").read_text().strip())

    git("commit","-m",f"[chatgpt-runner] result {job_id} rc={rc}",timeout=60)

    # Never force-push. If another commit landed, retry from a fresh remote
    # checkout on the next cycle.
    push = git("push","origin",f"HEAD:{branch}",check=False,timeout=90)
    if push.returncode != 0:
        raise RuntimeError(
            f"push failed for {job_id}; local spool retained\n"
            f"stdout:\n{push.stdout or ''}\nstderr:\n{push.stderr or ''}"
        )

    return git("rev-parse","HEAD").stdout.strip()

def wake(job_id, result_commit, cfg):
    wp = woken_path(job_id)
    if wp.exists():
        return True

    msg = (
        f"CONTINUE {job_id} "
        f"repo={cfg.get('repo_full_name') or 'unknown'} "
        f"branch={cfg['automation_branch']} "
        f"result_commit={result_commit}"
    )

    p = run(
        [cfg["wakeup_script"], msg],
        check=False,
        timeout=20,
        capture=False,
    )

    if p.returncode == 0:
        atomic_write(wp, result_commit + "\n")
        return True

    sys.stderr.write(
        f"wake failed for {job_id} rc={p.returncode}\n"
        f"{p.stdout or ''}{p.stderr or ''}\n"
    )
    return False

def upload_pending_spools(cfg):
    """Publish every completed local spool, but never wake from this phase.

    Result publication must not block fresh queued execution merely because an
    older ChatGPT wake attempt failed.
    """
    SPOOL.mkdir(parents=True, exist_ok=True)

    for sd in sorted(SPOOL.iterdir()):
        if not sd.is_dir() or not (sd / "COMPLETE").exists():
            continue

        job_id = sd.name

        try:
            result_commit = push_spool(job_id, cfg)
        except Exception:
            # GitHub/network/merge problems are retryable. Spool remains intact.
            traceback.print_exc()
            continue

        print(
            f"[{now()}] confirmed GitHub result {job_id} commit={result_commit}",
            flush=True
        )


def wake_one_pending_result(cfg):
    """Attempt at most one undelivered wake per cycle.

    Fresh job execution happens before this function is called. This prevents
    stale wake retries from starving the queue.
    """
    SPOOL.mkdir(parents=True, exist_ok=True)

    for sd in sorted(SPOOL.iterdir()):
        if not sd.is_dir() or not (sd / "COMPLETE").exists():
            continue

        job_id = sd.name

        if woken_path(job_id).exists():
            continue

        try:
            result_commit = push_spool(job_id, cfg)
        except Exception:
            traceback.print_exc()
            continue

        print(
            f"[{now()}] wake attempt {job_id} commit={result_commit}",
            flush=True
        )

        wake(job_id, result_commit, cfg)

        # Only one wake attempt per poll cycle.
        return
