needles = sys.argv[2:]

sqlite_files = []

for p in base.rglob("*"):
    try:
        if not p.is_file():
            continue
        with p.open("rb") as f:
            if f.read(16) == b"SQLite format 3\x00":
                sqlite_files.append(p)
    except Exception:
        pass

print(f"SQLite databases found: {len(sqlite_files)}")

def qi(name):
    return '"' + name.replace('"', '""') + '"'

for db in sqlite_files:
    print()
    print("=" * 78)
    print("DATABASE:", db)
    print("=" * 78)

    try:
        con = sqlite3.connect(
            f"file:{db}?mode=ro",
            uri=True,
            timeout=2,
        )
        con.row_factory = sqlite3.Row
    except Exception as e:
        print("OPEN ERROR:", repr(e))
        continue

    try:
        tables = [
            row[0]
            for row in con.execute(
                "SELECT name FROM sqlite_master "
                "WHERE type='table' AND name NOT LIKE 'sqlite_%' "
                "ORDER BY name"
            )
        ]
        print("TABLES:", ", ".join(tables))

        print()
        print("--- table counts ---")
        for table in tables:
            try:
                n = con.execute(
                    f"SELECT COUNT(*) FROM {qi(table)}"
                ).fetchone()[0]
                print(f"{table}: {n}")
            except Exception as e:
                print(f"{table}: COUNT ERROR {e!r}")

        print()
        print("--- relevant schemas ---")
        for table in tables:
            low = table.lower()
            if any(k in low for k in (
                "session", "message", "part", "task",
                "event", "error", "permission"
            )):
                try:
                    schema = con.execute(
                        "SELECT sql FROM sqlite_master "
                        "WHERE type='table' AND name=?",
                        (table,),
                    ).fetchone()
                    print(f"\n[{table}]")
                    print(schema[0] if schema else "(no schema)")
                except Exception as e:
                    print(f"{table}: SCHEMA ERROR {e!r}")

        print()
        print("--- rows containing relevant IDs/markers ---")

        search_needles = list(needles) + [
            "CANARY_FIRST_RESPONSE_INVALID",
            "Summarize the task tool output above",
            "corrective",
            "task-splitter",
        ]

        for table in tables:
            try:
                cols = con.execute(
                    f"PRAGMA table_info({qi(table)})"
                ).fetchall()
            except Exception:
                continue

            text_cols = []
            for c in cols:
                name = c[1]
                typ = (c[2] or "").upper()
                # SQLite frequently stores JSON/text in weakly typed columns.
                if (
                    "CHAR" in typ
                    or "TEXT" in typ
                    or "CLOB" in typ
                    or typ == ""
                ):
                    text_cols.append(name)

            if not text_cols:
                continue

            conditions = []
            params = []

            for col in text_cols:
                for needle in search_needles:
                    conditions.append(
                        f"CAST({qi(col)} AS TEXT) LIKE ?"
                    )
                    params.append(f"%{needle}%")

            if not conditions:
                continue

            sql = (
                f"SELECT rowid AS __rowid__, * "
                f"FROM {qi(table)} "
                f"WHERE {' OR '.join(conditions)} "
                f"LIMIT 200"
            )

            try:
                rows = con.execute(sql, params).fetchall()
            except Exception as e:
                print(f"\n[{table}] QUERY ERROR: {e!r}")
                continue

            if not rows:
                continue

            print()
            print(f"### TABLE {table}: {len(rows)} matching row(s)")

            for row in rows:
                d = dict(row)
                print("---")
                for k, v in d.items():
                    s = repr(v)
                    if len(s) > 8000:
                        s = s[:8000] + "...<TRUNCATED>"
                    print(f"{k}={s}")

    except Exception as e:
        print("DATABASE ERROR:", repr(e))
    finally:
        try:
            con.close()
        except Exception:
            pass
